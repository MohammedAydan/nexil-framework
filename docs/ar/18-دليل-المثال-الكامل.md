## 4. مكون تفاعلي صغير

```tsx
// src/components/QuantityPicker.tsx
import { state } from '@nexil/reactivity'

export function QuantityPicker() {
  const quantity = state(1)
  return (
    <div aria-label="الكمية">
      <button
        type="button"
        onClick$={() => {
          quantity.set((previous) => Math.max(1, previous - 1))
        }}
      >
        −
      </button>
      <output aria-live="polite">{quantity()}</output>
      <button
        type="button"
        onClick$={() => {
          quantity.set((previous) => previous + 1)
        }}
      >
        +
      </button>
    </div>
  )
}
```

هذا المكون وحده يحتاج chunk؛ ProductCard لا يحتاجه.

## 5. Route مع SEO

```tsx
// src/routes/index.tsx
import { renderHead } from '@nexil/seo'
import { ProductCard } from '../components/ProductCard.js'
import { QuantityPicker } from '../components/QuantityPicker.js'
import { getFeaturedProduct } from '../lib/server/products.js'

export default async function Home() {
  const product = await getFeaturedProduct()
  return (
    <>
      <head>
        {renderHead({
          title: `${product.name} | متجري`,
          description: 'منتج عربي مبني بصفحة HTML أولًا باستخدام Nexil.',
          canonical: 'https://example.com/',
          image: 'https://example.com/og/home.png',
          ogType: 'product',
        })}
      </head>
      <main dir="rtl">
        <h1>منتج الأسبوع</h1>
        <ProductCard product={product} />
        <QuantityPicker />
        <form action="/__nexil/actions/home/subscribe" method="post">
          <label for="email">اشترك في التحديثات</label>
          <input id="email" name="email" type="email" required />
          <button type="submit">اشتراك</button>
        </form>
      </main>
    </>
  )
}
```

## 6. Action

```ts
// src/routes/subscribe.action.ts
export async function subscribe(request: Request) {
  const form = await request.formData()
  const email = String(form.get('email') ?? '').trim()
  if (!email.includes('@')) {
    return { ok: false, errors: [{ field: 'email', message: 'البريد غير صالح' }] }
  }
  // أرسل إلى queue أو database بعد authorization وrate limit.
  return { ok: true, data: { queued: true } }
}
```

يجب ربط اسم Action بالـ route convention المعتمد في مشروعك. لا تعتبر هذا المثال نظام اشتراك إنتاجيًا من دون rate limit وdouble opt-in وstorage آمن.

## 9. ما الذي يمكن تحسينه؟

في تطبيق حقيقي أضف database adapter، session/auth، durable idempotency، image pipeline كامل، error boundary، loading state، observability مع consent، وCI gate يمنع merge عند تجاوز ميزانية العميل.
